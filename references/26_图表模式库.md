## 1. 评价与排名类

### 排序条形 + TOP-N 高亮（综合评价主图）

```python
def plot_ranking(ax, names, scores, highlight_n=3):
    """横向排序条形，前 N 名用主色高亮，其余灰；数值标在条尾。"""
    idx = np.argsort(scores)[::-1]
    names = [names[i] for i in idx]; scores = [scores[i] for i in idx]
    colors = [PALETTE["primary"] if i < highlight_n else PALETTE["baseline"]
              for i in range(len(names))]
    y = range(len(names))
    ax.barh(y, scores, color=colors, edgecolor="white", linewidth=0.5, height=0.65)
    ax.set_yticks(y); ax.set_yticklabels(names); ax.invert_yaxis()
    ax.set_xlabel("综合得分（分）")
    for i in range(min(highlight_n, len(scores))):
        ax.text(scores[i] + 0.01, i, f"{scores[i]:.4f}", va="center", fontsize=8)
    ax.spines["right"].set_visible(False); ax.spines["top"].set_visible(False)
```

## 2. 预测类

```python
def plot_forecast(ax, hist_t, hist_v, fut_t, fut_mean, fut_lo=None, fut_hi=None):
    """历史序列+预测均值+置信带；竖虚线分隔历史与预测。"""
    ax.plot(hist_t, hist_v, color=PALETTE["primary"], linewidth=2,
            marker="o", markersize=4, label="历史观测")
    ax.plot(fut_t, fut_mean, color=PALETTE["positive"], linewidth=2,
            marker="s", markersize=4, label="预测值")
    if fut_lo is not None and fut_hi is not None:
        ax.fill_between(fut_t, fut_lo, fut_hi, color=PALETTE["positive"],
                        alpha=0.15, label="95%预测区间")
    ax.axvline(x=hist_t[-1], color=PALETTE["baseline"], linestyle=":",
               linewidth=1, alpha=0.7)
    ax.set_xlabel("日期（周）"); ax.set_ylabel("销量（千克）")
    ax.legend(); ax.spines["right"].set_visible(False); ax.spines["top"].set_visible(False)
```

### 残差诊断图（模型检验配图，Type-1 诊断图一般进附录/检验小节）

```python
def plot_residuals(ax, y_pred, y_true):
    """残差 vs 预测值：应无系统性喇叭形/弯曲，否则模型设定有问题。"""
    res = np.asarray(y_true) - np.asarray(y_pred)
    ax.scatter(y_pred, res, c=PALETTE["primary"], s=25, alpha=0.6,
               edgecolors="white", linewidth=0.5)
    ax.axhline(0, color=PALETTE["baseline"], linestyle="--", linewidth=1)
    ax.set_xlabel("预测值"); ax.set_ylabel("残差")
    ax.spines["right"].set_visible(False); ax.spines["top"].set_visible(False)
```

```python
def plot_allocation_stacked(ax, categories, alloc_dict):
    """alloc_dict: {"品类A": 数组, "品类B": 数组, ...} 各层堆叠。"""
    bottom = np.zeros(len(categories))
    colors = plt.cm.viridis(np.linspace(0.15, 0.85, len(alloc_dict)))
    for (label, values), c in zip(alloc_dict.items(), colors):
        ax.bar(categories, values, bottom=bottom, label=label,
               color=c, edgecolor="white", linewidth=0.5)
        bottom += np.array(values)
    ax.set_xlabel("销售日期"); ax.set_ylabel("补货总量（千克）")
    ax.legend(fontsize=8); ax.spines["right"].set_visible(False); ax.spines["top"].set_visible(False)
```

### 基线 vs 主模型成对条形 + 提升百分比标注（对齐 10 号"每核心结果须与基线对比"）

```python
def plot_baseline_comparison(ax, metrics, baseline_vals, main_vals):
    x = np.arange(len(metrics)); width = 0.35
    ax.bar(x - width/2, baseline_vals, width, label="基线模型",
           color=PALETTE["baseline"], edgecolor="white", linewidth=0.5)
    ax.bar(x + width/2, main_vals, width, label="本文模型",
           color=PALETTE["primary"], edgecolor="white", linewidth=0.5)
    for i, (b, m) in enumerate(zip(baseline_vals, main_vals)):
        if m != b and b != 0:
            pct = (m - b) / abs(b) * 100
            c = PALETTE["positive"] if pct > 0 else PALETTE["negative"]
            ax.annotate(f"{'↑' if pct > 0 else '↓'}{abs(pct):.0f}%",
                        (x[i], max(b, m)), textcoords="offset points",
                        xytext=(0, 4), ha="center", fontsize=8, color=c)
    ax.set_xticks(x); ax.set_xticklabels(metrics)
    ax.set_ylabel("指标值"); ax.legend(fontsize=8)
    ax.spines["right"].set_visible(False); ax.spines["top"].set_visible(False)
```

```python
def plot_weight_sensitivity(ax, perturb_levels, rank_matrix, labels, top_n=5):
    """rank_matrix[i, j] = 第 i 档扰动下第 j 方案的排名；排名线越平越稳健。"""
    colors = plt.cm.tab10(np.linspace(0, 0.5, top_n))
    for j in range(min(top_n, rank_matrix.shape[1])):
        ax.plot(perturb_levels, rank_matrix[:, j], color=colors[j],
                linewidth=2, marker="o", markersize=5, label=labels[j])
    ax.set_xlabel("权重扰动幅度"); ax.set_ylabel("排名")
    ax.invert_yaxis(); ax.legend(fontsize=8)
    ax.spines["right"].set_visible(False); ax.spines["top"].set_visible(False)
```

```python
def plot_error_comparison(ax, model_names, metrics_dict):
    x = np.arange(len(model_names)); n = len(metrics_dict)
    width = 0.8 / n
    colors = plt.cm.viridis(np.linspace(0.15, 0.85, n))
    for i, (metric, vals) in enumerate(metrics_dict.items()):
        offset = (i - (n - 1) / 2) * width
        ax.bar(x + offset, vals, width, label=metric,
               color=colors[i], edgecolor="white", linewidth=0.5)
    ax.set_xticks(x); ax.set_xticklabels(model_names)
    ax.set_ylabel("误差值"); ax.legend(fontsize=8)
```

```python
    im = ax.imshow(corr, cmap="RdBu_r", vmin=-1, vmax=1, aspect="auto")
```

## 6. 组图模板（唯一允许的形态：≤2 个子图）

```python
def make_two_panel(figsize=(12, 5), vertical=False):
    """1×2（或 2×1）双子图，(a)(b) 标注。仅当一张图确实说不清时使用。"""
    from matplotlib.gridspec import GridSpec
    fig = plt.figure(figsize=figsize)
    gs = GridSpec(2, 1, hspace=0.4) if vertical else GridSpec(1, 2, wspace=0.3)
    ax_a, ax_b = fig.add_subplot(gs[0]), fig.add_subplot(gs[1])
    for ax, lab in ((ax_a, "(a)"), (ax_b, "(b)")):
        ax.text(-0.08, 1.03, lab, transform=ax.transAxes,
                fontsize=12, fontweight="bold", va="bottom", ha="left")
    return fig, ax_a, ax_b
```
